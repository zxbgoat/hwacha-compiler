cmake_minimum_required(VERSION 3.20)
project(hwacha-cc CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
include_directories(${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
add_executable(hwacha-cc src/main.cpp src/Analysis.cpp src/CodeGen.cpp src/GPUAdapt.cpp)
target_compile_definitions(hwacha-cc PRIVATE LLC_DEFAULT_PATH="${LLVM_TOOLS_BINARY_DIR}/llc")
llvm_map_components_to_libnames(LLVM_LIBS core support irreader analysis passes transformutils scalaropts instcombine)
target_link_libraries(hwacha-cc ${LLVM_LIBS})
