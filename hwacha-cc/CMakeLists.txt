cmake_minimum_required(VERSION 3.20)
project(hwacha-cc CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
include_directories(${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
add_executable(hwacha-cc src/main.cpp src/Analysis.cpp src/CodeGen.cpp src/GPUAdapt.cpp)
target_compile_definitions(hwacha-cc PRIVATE LLC_DEFAULT_PATH="${LLVM_TOOLS_BINARY_DIR}/llc")
llvm_map_components_to_libnames(LLVM_LIBS core support irreader linker analysis passes transformutils scalaropts instcombine)
target_link_libraries(hwacha-cc ${LLVM_LIBS})

# MLIR front end (optional: needs the conda `mlir` package of the same version as llvmdev)
find_package(MLIR QUIET CONFIG)
if(MLIR_FOUND)
  message(STATUS "Using MLIR ${MLIR_VERSION} from ${MLIR_DIR}: building hwacha-mlir")
  include_directories(${MLIR_INCLUDE_DIRS})
  add_executable(hwacha-mlir mlir/hwacha-mlir.cpp)
  target_link_libraries(hwacha-mlir MLIR ${LLVM_LIBS})
else()
  message(STATUS "MLIR not found: hwacha-mlir is not built")
endif()
